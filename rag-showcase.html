# graph_traversal.py — Expansión en 2 saltos mediante SQLite CTE
def traverse_legal_subgraph(cursor, seed_node: str, max_depth: int = 2):
    query = """
    WITH RECURSIVE legal_traversal(source, target, relation, depth, path) AS (
        SELECT source_node, target_node, relation_type, 1, 
               source_node || '->' || target_node
        FROM graph_edges WHERE source_node = :seed
        UNION ALL
        SELECT e.source_node, e.target_node, e.relation_type, t.depth + 1,
               t.path || '->' || e.target_node
        FROM graph_edges e
        JOIN legal_traversal t ON e.source_node = t.target
        WHERE t.depth < :max_depth AND instr(t.path, e.target_node) = 0
    )
    SELECT t.source, t.target, t.relation, t.depth, n.title, n.vigencia_status
    FROM legal_traversal t
    JOIN graph_nodes n ON t.target = n.node_id
    ORDER BY t.depth ASC, n.vigencia_status ASC;
    """
    return cursor.execute(query, {"seed": seed_node, "max_depth": max_depth}).fetchall()

# hybrid_retriever.py — Reciprocal Rank Fusion determinista
def reciprocal_rank_fusion(dense_ids: list[str], sparse_ids: list[str], k: int = 60):
    scores: dict[str, float] = {}
    
    for rank, doc_id in enumerate(dense_ids, start=1):
        scores[doc_id] = scores.get(doc_id, 0.0) + (1.0 / (k + rank))
        
    for rank, doc_id in enumerate(sparse_ids, start=1):
        scores[doc_id] = scores.get(doc_id, 0.0) + (1.0 / (k + rank))
        
    return sorted(scores.items(), key=lambda x: x[1], reverse=True)

Gating OCR Condicional: Evita ejecutar OCR en sentencias con texto digital seleccionable; solo invoca PaddleOCR a 300 DPI cuando la densidad de caracteres es menor a 80 por página, ahorrando más de 120 horas de GPU.

Pseudonimización BPE-Safe: Prohíbe reemplazar nombres por asteriscos o `[REDACTED]` (lo que distorsiona la distancia euclidiana y coseno); utiliza tokens sintácticos consistentes conformes con la Ley 29733.

# chunker_pseudonymizer.py — Fragmentación jerárquica BPE-safe
def parent_child_chunk(considerando_text: str, child_size: int = 512):
    # 1. Pseudonimización que no deforma tokenización
    sanitized = re.sub(r"\b\d{8}\b", "DOCUMENTO_IDENTIDAD_ANON", considerando_text)
    words = sanitized.split()
    chunks = []
    
    for i in range(0, len(words), child_size - 64):
        child = " ".join(words[i : i + child_size])
        chunks.append({
            "child_vector_text": child,
            "parent_full_text": considerando_text
        })
    return chunks
